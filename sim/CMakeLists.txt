cmake_minimum_required(VERSION 3.20)
project(mcporl_sim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Bit-exactness with Java requires strict IEEE semantics: no FMA contraction,
# no fast-math, SSE2 (not x87) doubles. These apply to every target.
add_library(mcp_strict_fp INTERFACE)
if(MSVC)
  target_compile_options(mcp_strict_fp INTERFACE /fp:strict)
else()
  target_compile_options(mcp_strict_fp INTERFACE -ffp-contract=off -fno-fast-math)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(i.86|x86)$")
    target_compile_options(mcp_strict_fp INTERFACE -msse2 -mfpmath=sse)
  endif()
endif()

add_library(mcp_sim INTERFACE)
target_include_directories(mcp_sim INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(mcp_sim INTERFACE mcp_strict_fp)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_latency tests/test_latency.cpp)
  target_link_libraries(test_latency PRIVATE mcp_sim)
  add_test(NAME latency COMMAND test_latency)

  # Java-exact math is checked against vectors generated by the JVM.
  find_package(Java COMPONENTS Runtime)
  if(Java_JAVA_EXECUTABLE)
    set(JREF_OUT ${CMAKE_CURRENT_BINARY_DIR}/jref)
    add_test(NAME jref_generate
             COMMAND ${Java_JAVA_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../oracle/jref/JRef.java ${JREF_OUT})
    set_tests_properties(jref_generate PROPERTIES FIXTURES_SETUP jref)

    add_executable(test_jmath tests/test_jmath.cpp)
    target_link_libraries(test_jmath PRIVATE mcp_sim)
    add_test(NAME jmath COMMAND test_jmath ${JREF_OUT})
    set_tests_properties(jmath PROPERTIES FIXTURES_REQUIRED jref)
  else()
    message(WARNING "Java not found: skipping Java-parity math tests")
  endif()
endif()

cmake_minimum_required(VERSION 3.20)
project(mcporl_sim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Bit-exactness with Java requires strict IEEE semantics: no FMA contraction,
# no fast-math, SSE2 (not x87) doubles. These apply to every target.
add_library(mcp_strict_fp INTERFACE)
if(MSVC)
  # /fp:precise is value-safe and, since VS 2022, never contracts a*b+c into
  # an FMA unless /fp:contract is given. /fp:strict adds nothing we need and
  # blocks many optimisations. The bench_hash test verifies bit-exactness.
  target_compile_options(mcp_strict_fp INTERFACE /fp:precise)
else()
  target_compile_options(mcp_strict_fp INTERFACE -ffp-contract=off -fno-fast-math)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(i.86|x86)$")
    target_compile_options(mcp_strict_fp INTERFACE -msse2 -mfpmath=sse)
  endif()
endif()

add_library(mcp_sim INTERFACE)
target_include_directories(mcp_sim INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(mcp_sim INTERFACE mcp_strict_fp)

add_executable(replay tools/replay.cpp)
target_link_libraries(replay PRIVATE mcp_sim)

add_executable(duel_replay tools/duel_replay.cpp)
target_link_libraries(duel_replay PRIVATE mcp_sim)

add_executable(bench_tick tools/bench_tick.cpp)
target_link_libraries(bench_tick PRIVATE mcp_sim)

add_executable(bench_duel tools/bench_duel.cpp)
target_link_libraries(bench_duel PRIVATE mcp_sim)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_latency tests/test_latency.cpp)
  target_link_libraries(test_latency PRIVATE mcp_sim)
  add_test(NAME latency COMMAND test_latency)

  add_test(NAME bench_hash
           COMMAND ${CMAKE_COMMAND} -DBENCH=$<TARGET_FILE:bench_tick>
             -DSIN=${CMAKE_CURRENT_SOURCE_DIR}/data/sin_table.bin -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/bench_hash.cmake)

  add_executable(test_pick tests/test_pick.cpp)
  target_link_libraries(test_pick PRIVATE mcp_sim)
  add_test(NAME pick COMMAND test_pick ${CMAKE_CURRENT_SOURCE_DIR}/data/sin_table.bin
           ${CMAKE_CURRENT_SOURCE_DIR}/data/pick_golden.txt)

  add_executable(test_lpvec3 tests/test_lpvec3.cpp)
  target_link_libraries(test_lpvec3 PRIVATE mcp_sim)
  add_test(NAME lpvec3 COMMAND test_lpvec3 ${CMAKE_CURRENT_SOURCE_DIR}/data/lpvec3_golden.txt)

  # Java-exact math is checked against vectors generated by the JVM.
  find_package(Java COMPONENTS Runtime)
  if(Java_JAVA_EXECUTABLE)
    set(JREF_OUT ${CMAKE_CURRENT_BINARY_DIR}/jref)
    add_test(NAME jref_generate
             COMMAND ${Java_JAVA_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../oracle/jref/JRef.java ${JREF_OUT})
    set_tests_properties(jref_generate PROPERTIES FIXTURES_SETUP jref)

    add_executable(test_jmath tests/test_jmath.cpp)
    target_link_libraries(test_jmath PRIVATE mcp_sim)
    add_test(NAME jmath COMMAND test_jmath ${JREF_OUT})
    set_tests_properties(jmath PROPERTIES FIXTURES_REQUIRED jref)

    # The committed sine table must be exactly what the JVM produces.
    add_test(NAME sin_table_matches_jvm
             COMMAND ${CMAKE_COMMAND} -E compare_files ${CMAKE_CURRENT_SOURCE_DIR}/data/sin_table.bin ${JREF_OUT}/sin_table.bin)
    set_tests_properties(sin_table_matches_jvm PROPERTIES FIXTURES_REQUIRED jref)
  else()
    message(WARNING "Java not found: skipping Java-parity math tests")
  endif()

  # Movement parity: every oracle scenario must replay bit-identically.
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    file(GLOB MCP_SCENARIOS ${CMAKE_CURRENT_SOURCE_DIR}/../oracle/scenarios/*.txt)
    foreach(scn ${MCP_SCENARIOS})
      get_filename_component(name ${scn} NAME_WE)
      add_test(NAME parity_${name}
               COMMAND ${CMAKE_COMMAND}
                 -DREPLAY=$<TARGET_FILE:replay> -DPYTHON=${Python3_EXECUTABLE}
                 -DSIN=${CMAKE_CURRENT_SOURCE_DIR}/data/sin_table.bin -DSCENARIO=${scn}
                 -DGOLDEN=${CMAKE_CURRENT_SOURCE_DIR}/../oracle/traces/${name}.jsonl
                 -DOUT=${CMAKE_CURRENT_BINARY_DIR}/sim_traces/${name}.jsonl
                 -DDIFF=${CMAKE_CURRENT_SOURCE_DIR}/../tools/tracediff.py
                 -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.cmake)
    endforeach()

    # Combat parity: every two-player oracle scenario, same rule.
    file(GLOB MCP_COMBAT_SCENARIOS ${CMAKE_CURRENT_SOURCE_DIR}/../oracle/combat/*.txt)
    foreach(scn ${MCP_COMBAT_SCENARIOS})
      get_filename_component(name ${scn} NAME_WE)
      add_test(NAME combat_${name}
               COMMAND ${CMAKE_COMMAND}
                 -DREPLAY=$<TARGET_FILE:duel_replay> -DPYTHON=${Python3_EXECUTABLE}
                 -DSIN=${CMAKE_CURRENT_SOURCE_DIR}/data/sin_table.bin -DSCENARIO=${scn}
                 -DGOLDEN=${CMAKE_CURRENT_SOURCE_DIR}/../oracle/traces/combat/${name}.jsonl
                 -DOUT=${CMAKE_CURRENT_BINARY_DIR}/sim_traces/combat/${name}.jsonl
                 -DDIFF=${CMAKE_CURRENT_SOURCE_DIR}/../tools/tracediff.py
                 -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.cmake)
    endforeach()
  endif()
endif()

# to the axe, hits the raised shield a few ticks later (weak, but the disable does

start A hotbar=netherite_sword,wooden_axe
start B z=3.0 yaw=180 offhand=shield
20 idle | idle
8  idle | use
1  slot=1 | use
3  idle | use
1  tap | use
1  slot=0 | use
14 idle | use
1  tap | use
20 idle | use
